# requirements-dev.txt
# Dev-only tooling this project's own ./lint.sh needs -- NOT part of
# requirements.txt deliberately: a real, running production install
# (see install/install.sh) has no use for a linter, type checker,
# dead-code scanner, or import-boundary checker. Install alongside
# requirements.txt for local development:
#   pip install -r requirements.txt -r requirements-dev.txt
#
# No version pins -- unlike requirements.txt's own starlette floor
# (a real, disclosed security fix, load-bearing), none of these four
# have a known compatibility or security concern that would justify
# pinning here.
ruff            # lint.sh -- style/correctness (see pyproject.toml's own [tool.ruff.lint] comment)
mypy            # lint.sh -- type checking (see pyproject.toml's own [tool.mypy] comment)
vulture         # lint.sh -- whole-program dead-code detection (see lint.sh's own top comment for why)
import-linter   # lint.sh -- architectural import-boundary enforcement (see pyproject.toml's own [tool.importlinter] section)
coverage        # AGENTS.md requires a coverage run on files you touch;
                # a documented command that needs an undeclared tool
                # is a rule nobody can follow
pytest          # test suite (tests/unit -- fast; tests/integration -- slow, needs Ollama)
httpx2          # test-only -- Starlette's TestClient imports it (as httpx) for
                # tests/integration/test_api.py. Not a typosquat despite the name:
                # starlette 1.x does `import httpx2 as httpx` internally.

# S3-compatible storage, to test that the mirror really works over
# object storage rather than only that the options are passed along.
# moto[server] specifically: the decorator patches boto3, which
# pyiceberg does not use -- it goes through PyArrow own S3 client and
# needs a real endpoint on a port.
moto[server]
boto3
flask-cors

# A REAL POSTGRESQL, as a wheel, for the test suite.
#
# WHY A TEST DEPENDENCY AND NOT A RUNTIME ONE: a deployment connects to
# a PostgreSQL somebody else runs. This exists so the adapter's tests
# run against a genuine server rather than a mock -- an untested
# database adapter would be the largest unverified thing in this
# codebase, and every other commit here is verified by measurement.
#
# IT IS NOT SMALL. That cost is accepted deliberately: the alternative
# was tests that skip wherever no PostgreSQL happens to be installed,
# which means the adapter is verified nowhere in particular.
#
# PYTHON 3.12 ONLY, which is why the marker is here. pgserver 0.1.4
# publishes a cp312 wheel and no cp313 one, so a 3.13 environment has
# nothing to install and `pip install -r` FAILS outright -- which it
# did, on a real machine, immediately after this line was added
# without the marker.
#
# THE ADAPTER'S TESTS SKIP WITHOUT IT, via importorskip. So a 3.13
# developer gets a working install and 19 fewer tests, rather than no
# install at all. That is the right way round: a dependency that
# cannot be satisfied must not block the ones that can.
pgserver<1.0; python_version < "3.13"
